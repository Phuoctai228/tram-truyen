<!-- Interactive script for table selection and filter feedback -->
<script>
    (function() {
      // Checkbox master toggle
      const masterCheckbox = document.querySelector('thead input[type="checkbox"]');
      const rowCheckboxes = document.querySelectorAll('tbody input[type="checkbox"]');

      if (masterCheckbox) {
        masterCheckbox.addEventListener('change', function() {
          rowCheckboxes.forEach(cb => {
            cb.checked = masterCheckbox.checked;
          });
        });
      }

      // Action toast simulation for quick interactions
      const actionButtons = document.querySelectorAll('tbody button');
      actionButtons.forEach(btn => {
        btn.addEventListener('click', function(e) {
          e.stopPropagation();
          const title = this.getAttribute('title') || 'Thao tác';
          console.info(`Kích hoạt: ${title}`);
        });
      });
    })();
  </script>
